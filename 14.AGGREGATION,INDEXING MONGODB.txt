db.orders.drop() 
db.orders.insertMany([ 
{cust:"A",amount:500,city:"Pune",status:"completed"}, 
{cust:"B",amount:300,city:"Mumbai",status:"pending"}, 
{cust:"A",amount:200,city:"Pune",status:"completed"}, 
{cust:"C",amount:700,city:"Delhi",status:"completed"}, 
{cust:"B",amount:400,city:"Mumbai",status:"completed"}, 
{cust:"D",amount:150,city:"Pune",status:"pending"} 
]) 
db.orders.aggregate([ 
{$group:{_id:"$cust",totalAmount:{$sum:"$amount"}}}, 
{$sort:{totalAmount:-1}} 
]) 
db.orders.aggregate([ 
{$match:{status:"completed"}}, 
{$group:{_id:"$city",avgAmount:{$avg:"$amount"}}}, 
{$project:{_id:0,city:"$_id",avgAmount:1}}, 
{$sort:{avgAmount:-1}} 
]) 
db.orders.aggregate([ 
{$match:{amount:{$gt:300}}}, 
{$group:{_id:"$status",total:{$sum:"$amount"}}} 
]) 
db.orders.createIndex({cust:1}) 
db.orders.createIndex({city:1,amount:-1}) 
db.orders.getIndexes() 
db.orders.find({cust:"A"}) 
db.orders.find({city:"Mumbai"}).sort({amount:-1})
