CREATE TABLE sales (
    id INT PRIMARY KEY,
    product_name VARCHAR(50),
    category VARCHAR(50),
    quantity INT,
    price DECIMAL(10,2)
);

SELECT * FROM sales;

SELECT category, COUNT(*) AS total_products
FROM sales
GROUP BY category;

SELECT category, SUM(quantity) AS total_quantity
FROM sales
GROUP BY category;

SELECT category, AVG(price) AS average_price
FROM sales
GROUP BY category;

SELECT category, SUM(quantity * price) AS total_sales
FROM sales
GROUP BY category;

SELECT category, SUM(quantity) AS total_quantity
FROM sales
GROUP BY category
HAVING SUM(quantity) > 10;

SELECT category, AVG(price) AS average_price
FROM sales
GROUP BY category
HAVING AVG(price) > 20000;
